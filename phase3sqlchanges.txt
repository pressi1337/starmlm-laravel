-- ============================================================
-- Customer ID for end-users — direct SQL (run if you can't use
-- `php artisan migrate`). Equivalent of the Laravel migration
-- 2026_05_28_000000_add_customer_id_to_users.
-- ============================================================

-- 1) Add the customer_id column. Unique so backfill / runtime collisions
--    fail loudly instead of producing duplicates.
ALTER TABLE `users`
    ADD COLUMN `customer_id` VARCHAR(20) NULL DEFAULT NULL AFTER `username`,
    ADD UNIQUE KEY `users_customer_id_unique` (`customer_id`);

-- 2) Backfill any existing role=2 users in id-asc order so the earliest
--    registered user becomes STARUP001, the next STARUP002, etc. Admins
--    and sub-admins (role 0/1) are left NULL.
--    Note: plain LPAD(n, 3, '0') would TRUNCATE values >= 1000 (LPAD shortens
--    when input length exceeds target). The CASE keeps zero-padding only up
--    to 999, then lets 1000+ flow through at their natural width.
SET @row := 0;
UPDATE `users`
SET `customer_id` = CONCAT(
    'STARUP',
    CASE
        WHEN (@row := @row + 1) < 10  THEN CONCAT('00', @row)
        WHEN @row < 100               THEN CONCAT('0',  @row)
        ELSE CAST(@row AS CHAR)
    END
)
WHERE `role` = 2 AND `customer_id` IS NULL
ORDER BY `id` ASC;

-- ============================================================
-- Backfill plan-product allocations for promoters who activated
-- BEFORE this feature. Equivalent of migration
-- 2026_06_17_000000_backfill_promoter_box_requests.
--
-- Every ACTIVATED promoter (status = 2) with no existing batch
-- gets one "Requested" batch for their level's full entitlement
-- (L0=1, L1=1, L2=10, L3=20, L4=30), reusing the delivery details
-- captured at activation. The NOT EXISTS guard makes it safe to
-- re-run and never duplicates a batch activation already created.
-- ============================================================

INSERT INTO `promoter_box_requests`
    (`user_id`, `user_promoter_id`, `level`, `quantity`,
     `delivery_type`, `delivery_address`, `pickup_date`, `contact_number`,
     `status`, `requested_at`, `created_by`, `updated_by`,
     `is_active`, `is_deleted`, `created_at`, `updated_at`)
SELECT
    up.`user_id`, up.`id`, up.`level`,
    CASE up.`level`
        WHEN 0 THEN 1 WHEN 1 THEN 1 WHEN 2 THEN 10
        WHEN 3 THEN 20 WHEN 4 THEN 30 ELSE 0
    END,
    up.`gift_delivery_type`, up.`gift_delivery_address`, up.`direct_pick_date`, up.`wh_number`,
    1, COALESCE(up.`activated_at`, up.`created_at`, NOW()), up.`user_id`, up.`user_id`,
    1, 0,
    COALESCE(up.`activated_at`, up.`created_at`, NOW()),
    COALESCE(up.`activated_at`, up.`created_at`, NOW())
FROM `user_promoters` up
WHERE up.`status` = 2
  AND up.`is_deleted` = 0
  AND up.`level` IN (0, 1, 2, 3, 4)
  AND NOT EXISTS (
      SELECT 1 FROM `promoter_box_requests` pbr
      WHERE pbr.`user_promoter_id` = up.`id` AND pbr.`is_deleted` = 0
  );

-- ============================================================
-- CUSTOMER ID NUMBERING — first block 1..500, then jump to 15130
--   STARUP001 .. STARUP500, then STARUP15130, STARUP15131, ...
--   Nothing is ever issued between 501 and 15129.
--
-- The generator lives in App\Models\User::nextCustomerId() (constants
-- CUSTOMER_ID_FIRST_BLOCK_END / CUSTOMER_ID_SECOND_BLOCK_START), so NEW
-- registrations follow this automatically once the code is deployed.
-- customer_id is only ever issued to end users (role = 2); admins and
-- sub-admins keep NULL.
--
-- The queries below are for CHECKING / FIXING existing rows.
-- ============================================================

-- 1) CURRENT STATE — how many members, highest number used, and whether any
--    row already sits inside the unused 501..15129 gap.
SELECT
    COUNT(*)                                                   AS total_members,
    MIN(CAST(SUBSTRING(customer_id, 7) AS UNSIGNED))           AS lowest_number,
    MAX(CAST(SUBSTRING(customer_id, 7) AS UNSIGNED))           AS highest_number,
    SUM(CAST(SUBSTRING(customer_id, 7) AS UNSIGNED)
        BETWEEN 501 AND 15129)                                 AS rows_in_gap
FROM `users`
WHERE `role` = 2 AND `customer_id` LIKE 'STARUP%';

-- 2) If rows_in_gap > 0, list them before changing anything.
SELECT `id`, `username`, `customer_id`, `created_at`
FROM `users`
WHERE `role` = 2
  AND `customer_id` LIKE 'STARUP%'
  AND CAST(SUBSTRING(`customer_id`, 7) AS UNSIGNED) BETWEEN 501 AND 15129
ORDER BY `id` ASC;

-- 3) OPTIONAL — renumber those rows into the 15130+ block, oldest first.
--    WARNING: this CHANGES the customer ID of existing members. Skip it if
--    they have already been told their number. Take a backup first.
SET @n := 15129;
UPDATE `users`
SET `customer_id` = CONCAT('STARUP', (@n := @n + 1))
WHERE `role` = 2
  AND `customer_id` LIKE 'STARUP%'
  AND CAST(SUBSTRING(`customer_id`, 7) AS UNSIGNED) BETWEEN 501 AND 15129
ORDER BY `id` ASC;

-- ============================================================
-- BACKFILL — invoices for batches already delivered
--
-- Delivered before the invoice feature existed, these rows have no price,
-- no financial year and no invoice number, so they show no Download button.
-- This fills all three.
--
-- BEFORE RUNNING:
--   1. Run the three migrations (bill_templates, invoice fields,
--      product_prices) — these columns must exist.
--   2. Fill in Admin -> Product Price. Step 1 copies FROM that master; with
--      no prices saved it fills nothing.
--   3. Take a backup. Every statement below is an UPDATE on live rows.
--
-- Run the steps IN ORDER. Each is safe to re-run: they only touch rows that
-- are still NULL, so a half-finished run can simply be repeated.
-- ============================================================

-- ---------- STEP 0: preview. Changes nothing. ----------
-- How many rows are waiting, and do they all have a price to copy?
SELECT b.`level`,
       COUNT(*)                                   AS delivered_without_invoice,
       MAX(p.`price`)                             AS price_that_will_apply,
       IF(MAX(p.`id`) IS NULL, 'NO PRICE SET', 'ok') AS status
FROM `promoter_box_requests` b
LEFT JOIN `product_prices` p
       ON p.`level` = b.`level` AND p.`is_deleted` = 0 AND p.`is_active` = 1
WHERE b.`status` = 3 AND b.`is_deleted` = 0 AND b.`invoice_no` IS NULL
GROUP BY b.`level`
ORDER BY b.`level`;
-- Any row reading "NO PRICE SET" will be SKIPPED. Set that level's price
-- first if you want those invoices raised.


-- ---------- STEP 1: copy the price from the master ----------
-- Snapshot, exactly as a live dispatch does: later edits to Product Price
-- will not change these rows.
UPDATE `promoter_box_requests` b
JOIN `product_prices` p
  ON p.`level` = b.`level` AND p.`is_deleted` = 0 AND p.`is_active` = 1
SET b.`rate_per_qty` = p.`price`,
    b.`mrp`          = p.`mrp`
WHERE b.`status` = 3
  AND b.`is_deleted` = 0
  AND b.`rate_per_qty` IS NULL;


-- ---------- STEP 2: financial year from the delivery date ----------
-- Indian FY starts 1 April, so a January delivery belongs to the year that
-- began the previous April. Matches InvoiceBuilder::financialYear() exactly.
UPDATE `promoter_box_requests`
SET `invoice_fy` = CONCAT(
        RIGHT(YEAR(`delivered_at`) - IF(MONTH(`delivered_at`) >= 4, 0, 1), 2),
        '-',
        RIGHT(YEAR(`delivered_at`) - IF(MONTH(`delivered_at`) >= 4, 0, 1) + 1, 2)
    )
WHERE `status` = 3
  AND `is_deleted` = 0
  AND `delivered_at` IS NOT NULL
  AND `rate_per_qty` IS NOT NULL
  AND `invoice_fy` IS NULL;


-- ---------- STEP 3: sequential number within each year ----------
-- Numbered in delivery order, continuing after any number already issued in
-- that year, so nothing is duplicated or renumbered.
-- (MySQL 8+. A temp table is used rather than a derived table in the UPDATE,
--  because MySQL refuses to read the same table it is updating inline.)
CREATE TEMPORARY TABLE `tmp_invoice_seq` AS
SELECT b.`id`,
       COALESCE(m.`maxno`, 0)
         + ROW_NUMBER() OVER (PARTITION BY b.`invoice_fy`
                              ORDER BY b.`delivered_at`, b.`id`) AS `new_no`
FROM `promoter_box_requests` b
LEFT JOIN (
    SELECT `invoice_fy` AS `fy`, MAX(`invoice_no`) AS `maxno`
    FROM `promoter_box_requests`
    WHERE `invoice_no` IS NOT NULL
    GROUP BY `invoice_fy`
) m ON m.`fy` = b.`invoice_fy`
WHERE b.`status` = 3
  AND b.`is_deleted` = 0
  AND b.`invoice_fy` IS NOT NULL
  AND b.`rate_per_qty` IS NOT NULL
  AND b.`invoice_no` IS NULL;

UPDATE `promoter_box_requests` b
JOIN `tmp_invoice_seq` t ON t.`id` = b.`id`
SET b.`invoice_no` = t.`new_no`;

DROP TEMPORARY TABLE `tmp_invoice_seq`;


-- ---------- STEP 4: verify ----------
-- Numbers per year: `issued` and `latest` should match, with no gaps.
SELECT `invoice_fy`, COUNT(*) AS issued,
       MIN(`invoice_no`) AS first_no, MAX(`invoice_no`) AS latest
FROM `promoter_box_requests`
WHERE `invoice_no` IS NOT NULL
GROUP BY `invoice_fy`
ORDER BY `invoice_fy`;

-- No duplicates within a year (must return zero rows):
SELECT `invoice_fy`, `invoice_no`, COUNT(*) AS c
FROM `promoter_box_requests`
WHERE `invoice_no` IS NOT NULL
GROUP BY `invoice_fy`, `invoice_no`
HAVING c > 1;

-- Anything still left without an invoice (expected: only levels with no
-- price set, and any row with no delivered_at):
SELECT `id`, `user_id`, `level`, `quantity`, `delivered_at`,
       `rate_per_qty`, `invoice_fy`, `invoice_no`
FROM `promoter_box_requests`
WHERE `status` = 3 AND `is_deleted` = 0 AND `invoice_no` IS NULL;
